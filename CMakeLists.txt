cmake_minimum_required(VERSION 3.24)
project(OneStep LANGUAGES C CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "OneStep requires Windows x64 and the MSVC toolchain. Use Developer PowerShell.")
endif()
enable_language(RC)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MINECRAFT_VERSION "1.21.11" CACHE STRING "Exact Fabric target")
set(_versions 1.21.11 26.1 26.1.1 26.1.2)
set(_ids 12111 26100 26101 26102)
list(FIND _versions "${MINECRAFT_VERSION}" _index)
if(_index EQUAL -1)
  message(FATAL_ERROR "Unsupported Minecraft version: ${MINECRAFT_VERSION}")
endif()
list(GET _ids ${_index} _target_id)
option(ONESTEP_LOCAL_TEST "Build isolated optimized Debug test DLL" OFF)
option(ONESTEP_PROTECTION_MAP "Emit linker map for optional protector" OFF)
if(ONESTEP_LOCAL_TEST)
  if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    message(FATAL_ERROR "ONESTEP_LOCAL_TEST requires Debug configuration.")
  endif()
  # MSVC's default Debug /RTC1 cannot be combined with the test DLL's /O2.
  foreach(_language C CXX)
    string(REGEX REPLACE "/RTC[^ ]*|/Od|/Ob0" "" CMAKE_${_language}_FLAGS_DEBUG "${CMAKE_${_language}_FLAGS_DEBUG}")
  endforeach()
endif()

# Read the existing project manifest; source lists cannot silently diverge.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/OneStep.vcxproj")
file(STRINGS "${CMAKE_SOURCE_DIR}/OneStep.vcxproj" _entries REGEX "<ClCompile Include=")
set(_sources)
set(_vendor)
foreach(_entry IN LISTS _entries)
  string(REGEX REPLACE ".*<ClCompile Include=\"([^\"]+)\".*" "\\1" _pattern "${_entry}")
  string(REPLACE "\\" "/" _pattern "${_pattern}")
  file(GLOB _expanded CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/${_pattern}")
  list(APPEND _sources ${_expanded})
  if(_pattern MATCHES "^include/")
    list(APPEND _vendor ${_expanded})
  endif()
endforeach()

# All generated data lives in this build's directory, not the source tree.
set(_generated "${CMAKE_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${_generated}")
set(_bridge_dir "${_generated}/assets/hybrid")
set(_java_sources)
set(_java_outputs)
foreach(_bridge GameBridge UtilityBridge ItemBridge InventoryBridge WorldBridge ClientTickRelay)
  list(APPEND _java_sources "${CMAKE_SOURCE_DIR}/src/bridge/${_bridge}.java")
  list(APPEND _java_outputs "${_bridge_dir}/net/onestep/hybrid/${_bridge}.class")
endforeach()
find_package(Java REQUIRED COMPONENTS Development)
add_custom_command(OUTPUT ${_java_outputs}
  COMMAND "${CMAKE_COMMAND}" -E make_directory "${_bridge_dir}"
  COMMAND "${Java_JAVAC_EXECUTABLE}" --release 17 -encoding UTF-8 -g:none -d "${_bridge_dir}" ${_java_sources}
  DEPENDS ${_java_sources} VERBATIM)

set(_contract_stamp "${_generated}/contracts.stamp")
file(GLOB _bridge_cpp CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/services/game/*bridge.cpp")
add_custom_command(OUTPUT "${_contract_stamp}"
  COMMAND Python3::Interpreter "${CMAKE_SOURCE_DIR}/scripts/Check-GameBridgeContracts.py" --classes "${_bridge_dir}"
  COMMAND "${CMAKE_COMMAND}" -E touch "${_contract_stamp}"
  DEPENDS ${_java_outputs} ${_bridge_cpp} scripts/Check-GameBridgeContracts.py scripts/minecraft_classfile.py VERBATIM)

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/assets.rc")
file(READ "${CMAKE_SOURCE_DIR}/assets.rc" _rc)
string(REPLACE "\\\\" "/" _rc "${_rc}")
string(REPLACE "\"assets/" "\"${CMAKE_SOURCE_DIR}/assets/" _rc "${_rc}")
foreach(_output IN LISTS _java_outputs)
  get_filename_component(_class "${_output}" NAME)
  string(REPLACE "${CMAKE_SOURCE_DIR}/assets/hybrid/net/onestep/hybrid/${_class}" "${_output}" _rc "${_rc}")
endforeach()
file(CONFIGURE OUTPUT "${_generated}/assets.rc" CONTENT "${_rc}" @ONLY)
string(REGEX MATCHALL "RCDATA \"[^\"]+\"" _resource_entries "${_rc}")
set(_resource_files)
foreach(_resource IN LISTS _resource_entries)
  string(REGEX REPLACE "RCDATA \"([^\"]+)\"" "\\1" _path "${_resource}")
  list(APPEND _resource_files "${_path}")
endforeach()
set_source_files_properties("${_generated}/assets.rc" PROPERTIES OBJECT_DEPENDS "${_resource_files};${CMAKE_SOURCE_DIR}/resource.h;${_contract_stamp}")
add_custom_target(onestep_generated DEPENDS ${_java_outputs} "${_contract_stamp}")

add_library(Onestep SHARED ${_sources} "${_generated}/assets.rc")
add_dependencies(Onestep onestep_generated)
target_compile_features(Onestep PRIVATE cxx_std_20)
target_compile_definitions(Onestep PRIVATE ONESTEP_CMAKE_BUILD=1 ONESTEP_MINECRAFT_VERSION=${_target_id} NOMINMAX WIN32_LEAN_AND_MEAN _WINDOWS _USRDLL)
target_compile_options(Onestep PRIVATE /W3 /utf-8 /bigobj /sdl "$<$<COMPILE_LANGUAGE:CXX>:/permissive->")
target_include_directories(Onestep BEFORE PRIVATE "${_generated}" "${CMAKE_SOURCE_DIR}" src/modules include/jni include/jni/win32 include/imgui include/glm include/minhook include)
target_precompile_headers(Onestep PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:${CMAKE_SOURCE_DIR}/include/pch.hpp>")
set_source_files_properties(${_vendor} PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
target_link_libraries(Onestep PRIVATE "${CMAKE_SOURCE_DIR}/include/jni/jvm.lib" opengl32 windowscodecs)
target_link_options(Onestep PRIVATE /SUBSYSTEM:WINDOWS /OPT:REF /OPT:ICF /INCREMENTAL:NO)
if(ONESTEP_PROTECTION_MAP)
  target_link_options(Onestep PRIVATE /MAP)
endif()
set_property(TARGET Onestep PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_property(TARGET Onestep PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
if(ONESTEP_LOCAL_TEST)
  target_compile_definitions(Onestep PRIVATE ONESTEP_LOCAL_TEST_BUILD=1)
  target_compile_options(Onestep PRIVATE /O2 /Gy /GF)
  set_target_properties(Onestep PROPERTIES OUTPUT_NAME "Onestep-TEST-NO-LAUNCHER")
endif()
add_custom_command(TARGET Onestep POST_BUILD
  COMMAND Python3::Interpreter "${CMAKE_SOURCE_DIR}/scripts/Check-GameBridgeContracts.py" --classes "${_bridge_dir}" --dll "$<TARGET_FILE:Onestep>" VERBATIM)
